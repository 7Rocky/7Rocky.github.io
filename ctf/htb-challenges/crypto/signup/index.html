def repeating_xor_key(message, key):

    repeation = 1 + (len(message) // len(key))
    key = key * repeation
    key = key[:len(message)]
    
    msg = bytes([c ^ k for c, k in zip(message, key)])
    return msg

def domain_params_generation():

    q = getPrime(N)

    print(f"[+] q condition is satisfied : {N} bit")

    p = 0
    while not (isPrime(p) and len(bin(p)[2:]) == L):
        factor = getRandomRange(2**(L-N-1), 2**(L-N))
        p = factor*q + 1
        
    print(f"[+] p condition is satisfied : {L} bit")

        g = pow(h, factor, p)

    print(f"[+] g condition is satisfied ")

    return(p, q, g)


def key_generation(p, q, g):

def sign(message, private_key, parameters):

    p, q, g = parameters
    k = getRandomRange(1, q-1)

    r = 0
    while r == 0:
        r = pow(g, k, p) % q

    s = 0
    while s == 0:
        Hm = int(sha512(message.encode('utf-8')).hexdigest(), 16)
        s = (inverse(k, q) * (Hm + private_key*r)) % q

    return (r, s)


def verify(message, signature, public_key, parameters):

    p, q, g = parameters
    r, s = signature

    if 0 < r < q and 0 < s < q:
        
        w = inverse(s, q)
        Hm = int(sha512(message.encode('utf-8')).hexdigest(), 16)
        u1 = (Hm * w) % q
        u2 = (r * w) % q
        v = ( (pow(g, u1, p) * pow(public_key, u2, p)) % p ) % q

        if v == r:
            print(f"[+] Valid signature")
            return True
        else:
            print("[!] Invalid signature")
            return False

Tenemos los parámetros de una implementación del algoritmo de firma digital (DSA), una larga lista de mensajes y firmas y finalmente la flag cifrada. La flag está cifrada con XOR y la clave privada $x$ del DSA.

Implementación del DSA

El código para el DSA es correcto (más información aquí):

  • Genera números primos $p$ y $q$ (tales que $p = mq + 1$ para cierto entero $m$)
  • Luego valores públicos $g$ e $y$
  • Y valores privados $k$ y $x$

Para firmar mensajes, utiliza SHA512 como función de hash y calcula lo siguiente:

$$ r = g^k \mod{q} $$

$$ s = \left(\mathrm{SHA512}{(m)} + x \cdot r\right) \cdot k^{-1} \mod{q} $$

Y la firma es el par $(r, s)$.

def get_number(pattern: str, res: str) -> int:
    return int(re.findall(pattern, res)[0])

El fallo de seguridad

El DSA puede verse comprometido cuando los valores de los nonce no se eligen al azar o se usan más de una vez. Esta vez tenemos un total de 100 mensajes y firmas. Sin embargo, si enumeramos el número de valores de $r_i$ distintos, tenemos 99 (tenemos mucha suerte, ya que los nonces $k$ se toman de getRandomRange):

    print(f'{len(r_i) = }, {len(set(r_i)) = }')
    same_k = (-1, -1)

$ python3 solve.py
len(r_i) = 100, flen(set(r_i)) = 99

Como $r_i = g^{k_i} \mod{q}$, if $r_i = r_j$, entonces $k_i = k_j$, ya que $2 \leqslant k_i < q$. Entonces, hay un par que reutiliza el mismo nonce $k$. Vamos a encontrarlo:

    same_k = (-1, -1)

    for i, r in enumerate(r_i):
        if r in r_i[i + 1:]:
            same_k = (i, r_i.index(r, i + 1))
            break

    print(f'{same_k = }')

$ python3 solve.py
len(r_i) = 100, len(set(r_i)) = 99
same_k = (29, 74)

Entonces, tenemos un mensaje $m_{29}$ con firma $(r, s_{29})$ y un mensaje $m_{74}$ con firma $(r, s_{74})$, donde $r = g^k \mod{q}$.

Por lo tanto, tenemos

$$ \begin{cases} s_{29} = \left(\mathrm{SHA512}{(m_{29})} + x \cdot r\right) \cdot k^{-1} \mod{q} \\ s_{74} = \left(\mathrm{SHA512}{(m_{74})} + x \cdot r\right) \cdot k^{-1} \mod{q} \end{cases} $$

Si restamos ambas ecuaciones, obtenemos

$$ s_{29} - s_{74} = \left(\mathrm{SHA512}{(m_{29})} - \mathrm{SHA512}{(m_{74})}\right) \cdot k^{-1} \mod{q} $$

Y así podemos encontrar $k^{-1}$ como sigue

$$ k^{-1} = (s_{29} - s_{74}) \cdot \left(\mathrm{SHA512}{(m_{29})} - \mathrm{SHA512}{(m_{74})}\right)^{-1} \mod{q} $$

Luego $k = \left(k^{-1}\right)^{-1} \mod{q}$. Y una vez que tenemos $k$, somos capaces de obtener $x$:

$$ x = \left(s_{29} \cdot k - \mathrm{SHA512}{(m_{74})}\right) \cdot r^{-1} \mod{q} $$
