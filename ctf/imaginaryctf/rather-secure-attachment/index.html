def try_solve_rsa(p, e, n, c):
    q = n // p
    phi_n = (p - 1) * (q - 1)
    d = pow(e, -1, phi_n)
    m = bytes.fromhex(hex(pow(c, d, n))[2:])

    if b'ictf{' in m:
        print(m.decode())
